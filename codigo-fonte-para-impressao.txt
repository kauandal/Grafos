   409	    /**
   410	     * Algoritmo de Prim a partir de um vértice inicial.
   411	     *
   412	     * Prim é definido para grafo não dirigido. Em grafo dirigido o
   413	     * algoritmo é aplicado sobre o grafo subjacente, ignorando o sentido.
   414	     * Se o grafo for desconexo, o resultado é uma floresta geradora
   415	     * mínima, com uma árvore por componente.
   416	     */
   417	    public ResultadoAGM prim(String idInicial) {
   418	        if (vertices.isEmpty()) {
   419	            throw new IllegalStateException("O grafo não possui vértices.");
   420	        }
   421	
   422	        Vertice inicial = (idInicial == null) ? vertices.get(0) : buscarVertice(idInicial);
   423	        if (inicial == null) {
   424	            throw new IllegalArgumentException("Vértice inicial não existe: " + idInicial);
   425	        }
   426	
   427	        Map<Vertice, List<Aresta>> incidencia = mapaIncidencia(false);
   428	        Set<Vertice> visitados = new LinkedHashSet<>();
   429	        List<Aresta> arvore = new ArrayList<>();
   430	        double custo = 0.0;
   431	        int componentes = 0;
   432	
   433	        // O vértice escolhido é a raiz da primeira árvore; os demais
   434	        // servem de raiz apenas se o grafo for desconexo.
   435	        List<Vertice> raizes = new ArrayList<>();
   436	        raizes.add(inicial);
   437	        for (Vertice v : vertices) {
   438	            if (v != inicial) {
   439	                raizes.add(v);
   440	            }
   441	        }
   442	
   443	        for (Vertice raiz : raizes) {
   444	            if (visitados.contains(raiz)) {
   445	                continue;
   446	            }
   447	            componentes++;
   448	
   449	            PriorityQueue<Aresta> fila = new PriorityQueue<>(
   450	                    Comparator.comparingDouble(Aresta::getPeso).thenComparing(Aresta::getId));
   451	
   452	            expandir(raiz, visitados, incidencia, fila);
   453	
   454	            while (!fila.isEmpty()) {
   455	                Aresta a = fila.poll();
   456	
   457	                Vertice fora = null;
   458	                if (!visitados.contains(a.getOrigem())) {
   459	                    fora = a.getOrigem();
   460	                } else if (!visitados.contains(a.getDestino())) {
   461	                    fora = a.getDestino();
   462	                }
   463	
   464	                // Aresta com as duas extremidades já na árvore formaria ciclo.
   465	                if (fora == null) {
   466	                    continue;
   467	                }
   468	
   469	                arvore.add(a);
   470	                custo += a.getPeso();
   471	                expandir(fora, visitados, incidencia, fila);
   472	            }
   473	        }
   474	
   475	        return new ResultadoAGM(inicial, arvore, custo, componentes);
   476	    }

   478	    /** Marca o vértice como pertencente à árvore e enfileira suas arestas candidatas. */
   479	    private void expandir(Vertice v, Set<Vertice> visitados,
   480	                          Map<Vertice, List<Aresta>> incidencia,
   481	                          PriorityQueue<Aresta> fila) {
   482	        visitados.add(v);
   483	        for (Aresta a : incidencia.get(v)) {
   484	            if (!visitados.contains(a.oposto(v))) {
   485	                fila.add(a);
   486	            }
   487	        }
   488	    }

   495	     * Busca em profundidade com vértice de saída e de chegada.
   496	     * A busca é interrompida assim que o vértice de chegada é alcançado.
   497	     * Em grafo dirigido o sentido dos arcos é respeitado.
   498	     */
   499	    public ResultadoBusca buscaEmProfundidade(String idOrigem, String idDestino) {
   500	        Vertice origem = buscarVertice(idOrigem);
   501	        Vertice destino = buscarVertice(idDestino);
   502	
   503	        if (origem == null) {
   504	            throw new IllegalArgumentException("Vértice de saída não existe: " + idOrigem);
   505	        }
   506	        if (destino == null) {
   507	            throw new IllegalArgumentException("Vértice de chegada não existe: " + idDestino);
   508	        }
   509	
   510	        Map<Vertice, List<Aresta>> incidencia = mapaIncidencia(dirigido);
   511	
   512	        Set<Vertice> visitados = new LinkedHashSet<>();
   513	        List<Vertice> ordem = new ArrayList<>();
   514	        List<Aresta> arvore = new ArrayList<>();
   515	        Map<Vertice, Aresta> chegada = new HashMap<>();
   516	
   517	        boolean encontrou = explorar(origem, destino, incidencia, visitados, ordem, arvore, chegada);
   518	
   519	        List<Aresta> caminhoArestas = new ArrayList<>();
   520	        List<Vertice> caminhoVertices = new ArrayList<>();
   521	        double custoCaminho = 0.0;
   522	
   523	        if (encontrou) {
   524	            Vertice atual = destino;
   525	            caminhoVertices.add(atual);
   526	            while (chegada.containsKey(atual)) {
   527	                Aresta a = chegada.get(atual);
   528	                caminhoArestas.add(0, a);
   529	                custoCaminho += a.getPeso();
   530	                atual = a.oposto(atual);
   531	                caminhoVertices.add(0, atual);
   532	            }
   533	        }
   534	
   535	        return new ResultadoBusca(origem, destino, encontrou, ordem, arvore,
   536	                caminhoVertices, caminhoArestas, custoCaminho);

   539	    /**
   540	     * Percurso em profundidade com pilha explícita, o que evita estouro de
   541	     * pilha de recursão em grafos grandes. A ordem de visita é a mesma da
   542	     * versão recursiva: segue sempre pela primeira aresta ainda não explorada.
   543	     */
   544	    private boolean explorar(Vertice origem, Vertice destino,
   545	                             Map<Vertice, List<Aresta>> incidencia,
   546	                             Set<Vertice> visitados, List<Vertice> ordem,
   547	                             List<Aresta> arvore, Map<Vertice, Aresta> chegada) {
   548	
   549	        visitados.add(origem);
   550	        ordem.add(origem);
   551	
   552	        if (origem == destino) {
   553	            return true;
   554	        }
   555	
   556	        Deque<Vertice> pilhaVertices = new ArrayDeque<>();
   557	        Deque<Iterator<Aresta>> pilhaArestas = new ArrayDeque<>();
   558	
   559	        pilhaVertices.push(origem);
   560	        pilhaArestas.push(incidencia.get(origem).iterator());
   561	
   562	        while (!pilhaVertices.isEmpty()) {
   563	            Vertice atual = pilhaVertices.peek();
   564	            Iterator<Aresta> it = pilhaArestas.peek();
   565	            boolean avancou = false;
   566	
   567	            while (it.hasNext()) {
   568	                Aresta a = it.next();
   569	                Vertice proximo = a.oposto(atual);
   570	
   571	                if (visitados.contains(proximo)) {
   572	                    continue;
   573	                }
   574	
   575	                visitados.add(proximo);
   576	                ordem.add(proximo);
   577	                arvore.add(a);
   578	                chegada.put(proximo, a);
   579	
   580	                if (proximo == destino) {
   581	                    return true;
   582	                }
   583	
   584	                pilhaVertices.push(proximo);
   585	                pilhaArestas.push(incidencia.get(proximo).iterator());
   586	                avancou = true;
   587	                break;
   588	            }
   589	
   590	            if (!avancou) {
   591	                pilhaVertices.pop();
   592	                pilhaArestas.pop();
   593	            }
   594	        }
   595	
   596	        return false;
   597	    }

   603	    /**
   604	     * Matriz de alcançabilidade (fecho transitivo) pelo algoritmo de Roy,
   605	     * também conhecido como Roy-Warshall, aplicado sobre a matriz booleana
   606	     * de adjacência.
   607	     */
   608	    public boolean[][] matrizAlcancabilidade() {
   609	        int n = vertices.size();
   610	        boolean[][] r = new boolean[n][n];
   611	        Map<Vertice, Integer> indices = indices();
   612	
   613	        // Todo vértice alcança a si mesmo.
   614	        for (int i = 0; i < n; i++) {
   615	            r[i][i] = true;
   616	        }
   617	
   618	        for (Aresta a : arestas) {
   619	            int i = indices.get(a.getOrigem());
   620	            int j = indices.get(a.getDestino());
   621	            r[i][j] = true;
   622	            if (!dirigido) {
   623	                r[j][i] = true;
   624	            }
   625	        }
   626	
   627	        // Fecho transitivo: se i alcança k e k alcança j, então i alcança j.
   628	        for (int k = 0; k < n; k++) {
   629	            for (int i = 0; i < n; i++) {
   630	                if (!r[i][k]) {
   631	                    continue;
   632	                }
   633	                for (int j = 0; j < n; j++) {
   634	                    if (r[k][j]) {
   635	                        r[i][j] = true;
   636	                    }
   637	                }
   638	            }
   639	        }
   640	        return r;
   641	    }

   643	    /**
   644	     * Componentes conexas (grafo não dirigido) ou fortemente conexas
   645	     * (grafo dirigido): v e w pertencem ao mesmo conjunto quando
   646	     * v alcança w e w alcança v na matriz de alcançabilidade.
   647	     */
   648	    public List<List<Vertice>> royComponentes() {
   649	        boolean[][] r = matrizAlcancabilidade();
   650	        int n = vertices.size();
   651	        boolean[] agrupado = new boolean[n];
   652	        List<List<Vertice>> componentes = new ArrayList<>();
   653	
   654	        for (int i = 0; i < n; i++) {
   655	            if (agrupado[i]) {
   656	                continue;
   657	            }
   658	            List<Vertice> componente = new ArrayList<>();
   659	            for (int j = 0; j < n; j++) {
   660	                if (!agrupado[j] && r[i][j] && r[j][i]) {
   661	                    agrupado[j] = true;
   662	                    componente.add(vertices.get(j));
   663	                }
   664	            }
   665	            componentes.add(componente);
   666	        }
   667	        return componentes;
   668	    }

   157	    private void inserirVertice() {
   158	        String id = perguntar("Identificador do novo vértice:");
   159	        if (id == null) {
   160	            return;
   161	        }
   162	        try {
   163	            grafo.inserirVertice(id);
   164	            imprimir("Vértice inserido: " + id);
   165	        } catch (RuntimeException ex) {
   166	            erro(ex);
   167	        }
   168	    }

   170	    private void inserirAresta() {
   171	        String id = perguntar("Identificador da aresta/arco:");
   172	        if (id == null) {
   173	            return;
   174	        }
   175	        String origem = perguntar(grafo.isDirigido() ? "Vértice de origem:" : "Primeiro vértice:");
   176	        if (origem == null) {
   177	            return;
   178	        }
   179	        String destino = perguntar(grafo.isDirigido() ? "Vértice de destino:" : "Segundo vértice:");
   180	        if (destino == null) {
   181	            return;
   182	        }
   183	        String texto = perguntar("Peso (custo):");
   184	        if (texto == null) {
   185	            return;
   186	        }
   187	        try {
   188	            grafo.inserirAresta(id, origem, destino, Double.parseDouble(texto.replace(',', '.')));
   189	            imprimir("Ligação inserida: " + grafo.buscarAresta(id));
   190	        } catch (NumberFormatException ex) {
   191	            erro(new IllegalArgumentException("Peso inválido: " + texto));
   192	        } catch (RuntimeException ex) {
   193	            erro(ex);
   194	        }
   195	    }

   263	    private void prim() {
   264	        String inicial = perguntar("Vértice inicial do Prim:");
   265	        if (inicial == null) {
   266	            return;
   267	        }
   268	        try {
   269	            Grafo.ResultadoAGM agm = grafo.prim(inicial);
   270	
   271	            StringBuilder sb = new StringBuilder("===== PRIM: ÁRVORE GERADORA MÍNIMA =====\n");
   272	            sb.append("Vértice inicial: ").append(agm.getInicial().getId()).append('\n');
   273	            for (Aresta a : agm.getArestas()) {
   274	                sb.append("  ").append(a.getId()).append(": ").append(a.getOrigem().getId())
   275	                        .append(" -- ").append(a.getDestino().getId())
   276	                        .append("  peso ").append(a.getPesoFormatado()).append('\n');
   277	            }
   278	            sb.append("Arestas: ").append(agm.getArestas().size())
   279	                    .append("   CUSTO TOTAL: ").append(Aresta.formatarPeso(agm.getCusto()));
   280	            if (agm.isFloresta()) {
   281	                sb.append("\nGrafo desconexo: floresta com ").append(agm.getComponentes()).append(" árvores.");
   282	            }
   283	            imprimir(sb.toString());
   284	
   285	            painel.realcar(agm.getArestas(), idsDe(agm.getArestas()), null);
   286	        } catch (RuntimeException ex) {
   287	            erro(ex);
   288	        }
   289	    }

   291	    private void busca() {
   292	        String origem = perguntar("Vértice de saída:");
   293	        if (origem == null) {
   294	            return;
   295	        }
   296	        String destino = perguntar("Vértice de chegada:");
   297	        if (destino == null) {
   298	            return;
   299	        }
   300	        try {
   301	            Grafo.ResultadoBusca b = grafo.buscaEmProfundidade(origem, destino);
   302	
   303	            StringBuilder sb = new StringBuilder("===== BUSCA EM PROFUNDIDADE GUIADA =====\n");
   304	            sb.append("Saída: ").append(b.getOrigem().getId())
   305	                    .append("   Chegada: ").append(b.getDestino().getId()).append('\n');
   306	            sb.append("Ordem de visita: ").append(b.getOrdemVisita()).append('\n');
   307	            sb.append("Arestas da árvore:\n");
   308	            for (Aresta a : b.getArestasArvore()) {
   309	                sb.append("  ").append(a.getId()).append(": ").append(a.getOrigem().getId())
   310	                        .append(grafo.isDirigido() ? " -> " : " -- ").append(a.getDestino().getId())
   311	                        .append("  peso ").append(a.getPesoFormatado()).append('\n');
   312	            }
   313	            if (b.isEncontrou()) {
   314	                sb.append("Caminho: ").append(b.getCaminhoVertices())
   315	                        .append("   custo ").append(Aresta.formatarPeso(b.getCustoCaminho()));
   316	            } else {
   317	                sb.append(b.getDestino().getId()).append(" não é alcançável a partir de ")
   318	                        .append(b.getOrigem().getId()).append('.');
   319	            }
   320	            imprimir(sb.toString());
   321	
   322	            Map<String, Color> cores = new HashMap<>();
   323	            cores.put(b.getOrigem().getId(), new Color(0xA8, 0xE0, 0xB0));
   324	            cores.put(b.getDestino().getId(), new Color(0xF5, 0xB4, 0xA8));
   325	            painel.realcar(b.getArestasArvore(), idsDe(b.getCaminhoArestas()), cores);
   326	        } catch (RuntimeException ex) {
   327	            erro(ex);
   328	        }
   329	    }

   331	    private void roy() {
   332	        List<List<Vertice>> componentes = grafo.royComponentes();
   333	
   334	        StringBuilder sb = new StringBuilder("===== ROY: COMPONENTES ");
   335	        sb.append(grafo.isDirigido() ? "FORTEMENTE CONEXAS" : "CONEXAS").append(" =====\n");
   336	
   337	        Map<String, Color> cores = new HashMap<>();
   338	        for (int i = 0; i < componentes.size(); i++) {
   339	            sb.append("C").append(i + 1).append(" = ").append(componentes.get(i)).append('\n');
   340	            for (Vertice v : componentes.get(i)) {
   341	                cores.put(v.getId(), PainelGrafo.PALETA[i % PainelGrafo.PALETA.length]);
   342	            }
   343	        }
   344	        sb.append("Total de componentes: ").append(componentes.size());
   345	        imprimir(sb.toString());
   346	
   347	        painel.realcar(null, null, cores);
   348	    }
