   258	    /**
   259	     * Algoritmo de Prim a partir de um vértice inicial.
   260	     *
   261	     * Prim é definido para grafo não dirigido. Em grafo dirigido o
   262	     * algoritmo é aplicado sobre o grafo subjacente, ignorando o sentido.
   263	     * Se o grafo for desconexo, o resultado é uma floresta geradora
   264	     * mínima, com uma árvore por componente.
   265	     */
   266	    public ResultadoAGM prim(String idInicial) {
   267	        if (vertices.isEmpty()) {
   268	            throw new IllegalStateException("O grafo não possui vértices.");
   269	        }
   270	
   271	        Vertice inicial = (idInicial == null) ? vertices.get(0) : buscarVertice(idInicial);
   272	        if (inicial == null) {
   273	            throw new IllegalArgumentException("Vértice inicial não existe: " + idInicial);
   274	        }
   275	
   276	        Map<Vertice, List<Aresta>> incidencia = incidenciaSemSentido();
   277	        Set<Vertice> visitados = new HashSet<>();
   278	        List<Aresta> arvore = new ArrayList<>();
   279	        double custo = 0.0;
   280	        int componentes = 0;
   281	
   282	        // O vértice escolhido é a raiz da primeira árvore; os demais
   283	        // servem de raiz apenas se o grafo for desconexo.
   284	        List<Vertice> raizes = new ArrayList<>();
   285	        raizes.add(inicial);
   286	        for (Vertice v : vertices) {
   287	            if (v != inicial) {
   288	                raizes.add(v);
   289	            }
   290	        }
   291	
   292	        for (Vertice raiz : raizes) {
   293	            if (visitados.contains(raiz)) {
   294	                continue;
   295	            }
   296	            componentes++;
   297	
   298	            PriorityQueue<Aresta> fila = new PriorityQueue<>(
   299	                    Comparator.comparingDouble(Aresta::getPeso).thenComparing(Aresta::getId));
   300	
   301	            expandir(raiz, visitados, incidencia, fila);
   302	
   303	            while (!fila.isEmpty()) {
   304	                Aresta a = fila.poll();
   305	
   306	                Vertice fora = null;
   307	                if (!visitados.contains(a.getOrigem())) {
   308	                    fora = a.getOrigem();
   309	                } else if (!visitados.contains(a.getDestino())) {
   310	                    fora = a.getDestino();
   311	                }
   312	
   313	                // Aresta com as duas extremidades já na árvore formaria ciclo.
   314	                if (fora == null) {
   315	                    continue;
   316	                }
   317	
   318	                arvore.add(a);
   319	                custo += a.getPeso();
   320	                expandir(fora, visitados, incidencia, fila);
   321	            }
   322	        }
   323	
   324	        return new ResultadoAGM(inicial, arvore, custo, componentes);
   325	    }

   327	    /** Marca o vértice como pertencente à árvore e enfileira suas arestas candidatas. */
   328	    private void expandir(Vertice v, Set<Vertice> visitados,
   329	                          Map<Vertice, List<Aresta>> incidencia,
   330	                          PriorityQueue<Aresta> fila) {
   331	        visitados.add(v);
   332	        for (Aresta a : incidencia.get(v)) {
   333	            if (!visitados.contains(a.oposto(v))) {
   334	                fila.add(a);
   335	            }
   336	        }

   344	     * Busca em profundidade com vértice de saída e de chegada.
   345	     * A busca é interrompida assim que o vértice de chegada é alcançado.
   346	     * Em grafo dirigido o sentido dos arcos é respeitado.
   347	     */
   348	    public ResultadoBusca buscaEmProfundidade(String idOrigem, String idDestino) {
   349	        Vertice origem = buscarVertice(idOrigem);
   350	        Vertice destino = buscarVertice(idDestino);
   351	
   352	        if (origem == null) {
   353	            throw new IllegalArgumentException("Vértice de saída não existe: " + idOrigem);
   354	        }
   355	        if (destino == null) {
   356	            throw new IllegalArgumentException("Vértice de chegada não existe: " + idDestino);
   357	        }
   358	
   359	        Set<Vertice> visitados = new HashSet<>();
   360	        List<Vertice> ordem = new ArrayList<>();
   361	        List<Aresta> arvore = new ArrayList<>();
   362	        Map<Vertice, Aresta> chegada = new HashMap<>();
   363	
   364	        boolean encontrou = explorar(origem, destino, visitados, ordem, arvore, chegada);
   365	
   366	        List<Aresta> caminhoArestas = new ArrayList<>();
   367	        List<Vertice> caminhoVertices = new ArrayList<>();
   368	        double custoCaminho = 0.0;
   369	
   370	        if (encontrou) {
   371	            Vertice atual = destino;
   372	            caminhoVertices.add(atual);
   373	            while (chegada.containsKey(atual)) {
   374	                Aresta a = chegada.get(atual);
   375	                caminhoArestas.add(0, a);
   376	                custoCaminho += a.getPeso();
   377	                atual = a.oposto(atual);
   378	                caminhoVertices.add(0, atual);
   379	            }
   380	        }
   381	
   382	        return new ResultadoBusca(origem, destino, encontrou, ordem, arvore,
   383	                caminhoVertices, caminhoArestas, custoCaminho);
   384	    }

   386	    /**
   387	     * Percurso recursivo em profundidade sobre a lista de adjacência.
   388	     * Visita o vértice atual, e para cada aresta ainda não explorada desce
   389	     * no vizinho não visitado. Retorna verdadeiro assim que alcança o
   390	     * destino, o que interrompe a busca.
   391	     *
   392	     * Em grafo dirigido a lista de adjacência já contém apenas os arcos que
   393	     * saem do vértice, então o sentido é respeitado sem tratamento extra.
   394	     */
   395	    private boolean explorar(Vertice atual, Vertice destino,
   396	                             Set<Vertice> visitados, List<Vertice> ordem,
   397	                             List<Aresta> arvore, Map<Vertice, Aresta> chegada) {
   398	
   399	        visitados.add(atual);
   400	        ordem.add(atual);
   401	
   402	        if (atual == destino) {
   403	            return true;
   404	        }
   405	
   406	        for (Aresta a : atual.getAdjacencias()) {
   407	            Vertice proximo = a.oposto(atual);
   408	
   409	            if (visitados.contains(proximo)) {
   410	                continue;
   411	            }
   412	
   413	            // Aresta da árvore: foi por ela que o vizinho foi descoberto.
   414	            arvore.add(a);
   415	            chegada.put(proximo, a);
   416	
   417	            if (explorar(proximo, destino, visitados, ordem, arvore, chegada)) {
   418	                return true;
   419	            }
   420	        }
   421	
   422	        return false;
   423	    }

   429	    /**
   430	     * Matriz de alcançabilidade (fecho transitivo) pelo algoritmo de Roy,
   431	     * também conhecido como Roy-Warshall, aplicado sobre a matriz booleana
   432	     * de adjacência.
   433	     */
   434	    public boolean[][] matrizAlcancabilidade() {
   435	        int n = vertices.size();
   436	        boolean[][] r = new boolean[n][n];
   437	
   438	        // Todo vértice alcança a si mesmo.
   439	        for (int i = 0; i < n; i++) {
   440	            r[i][i] = true;
   441	        }
   442	
   443	        // A posição do vértice na lista já é o seu índice na matriz.
   444	        for (Aresta a : arestas) {
   445	            int i = vertices.indexOf(a.getOrigem());
   446	            int j = vertices.indexOf(a.getDestino());
   447	            r[i][j] = true;
   448	            if (!dirigido) {
   449	                r[j][i] = true;
   450	            }
   451	        }
   452	
   453	        // Fecho transitivo: se i alcança k e k alcança j, então i alcança j.
   454	        for (int k = 0; k < n; k++) {
   455	            for (int i = 0; i < n; i++) {
   456	                if (!r[i][k]) {
   457	                    continue;
   458	                }
   459	                for (int j = 0; j < n; j++) {
   460	                    if (r[k][j]) {
   461	                        r[i][j] = true;
   462	                    }
   463	                }
   464	            }
   465	        }
   466	        return r;
   467	    }
   468	
   469	    /**
   470	     * Componentes conexas (grafo não dirigido) ou fortemente conexas
   471	     * (grafo dirigido): v e w pertencem ao mesmo conjunto quando
   472	     * v alcança w e w alcança v na matriz de alcançabilidade.
   473	     */
   474	    public List<List<Vertice>> royComponentes() {
   475	        boolean[][] r = matrizAlcancabilidade();
   476	        int n = vertices.size();
   477	        boolean[] agrupado = new boolean[n];
   478	        List<List<Vertice>> componentes = new ArrayList<>();
   479	
   480	        for (int i = 0; i < n; i++) {
   481	            if (agrupado[i]) {
   482	                continue;
   483	            }
   484	            List<Vertice> componente = new ArrayList<>();
   485	            for (int j = 0; j < n; j++) {
   486	                if (!agrupado[j] && r[i][j] && r[j][i]) {
   487	                    agrupado[j] = true;
   488	                    componente.add(vertices.get(j));
   489	                }
   490	            }
   491	            componentes.add(componente);
   492	        }
   493	        return componentes;
   494	    }

    95	    private void posicionarEmCirculo() {
    96	        List<Vertice> lista = grafo.getVertices();
    97	        int n = lista.size();

   110	    private void desenharAresta(Graphics2D g2, Aresta a) {
   111	        Point p1 = posicoes.get(a.getOrigem().getId());
   112	        Point p2 = posicoes.get(a.getDestino().getId());
   113	
   114	        boolean destaque = destacadas.contains(a.getId());
   115	        g2.setColor(destaque ? COR_DESTAQUE : Color.DARK_GRAY);
   116	        g2.setStroke(new BasicStroke(destaque ? 3f : 1.5f));
   117	
   118	        String texto = a.getId() + " (" + a.getPesoFormatado() + ")";
   119	
   120	        if (a.isLaco()) {
   121	            g2.drawOval(p1.x - 18, p1.y - RAIO - 26, 36, 30);
   122	            escrever(g2, texto, p1.x, p1.y - RAIO - 32);
   123	            return;
   124	        }
   125	
   126	        double dx = p2.x - p1.x;
   127	        double dy = p2.y - p1.y;
   128	        double dist = Math.hypot(dx, dy);
   129	        double ux = dx / dist;
   130	        double uy = dy / dist;
   131	
   132	        // Em grafo dirigido a reta é deslocada para a lateral. Como a
   133	        // perpendicular acompanha o sentido origem -> destino, o arco de ida e
   134	        // o de volta entre o mesmo par caem em lados opostos automaticamente.
   135	        int desvio = grafo.isDirigido() ? 16 : 0;
   136	        int deslocX = (int) (-uy * desvio);
   137	        int deslocY = (int) (ux * desvio);
   138	
   139	        // A reta começa e termina na borda dos círculos, não no centro.
   140	        int x1 = (int) (p1.x + ux * RAIO) + deslocX;
   141	        int y1 = (int) (p1.y + uy * RAIO) + deslocY;
   142	        int x2 = (int) (p2.x - ux * RAIO) + deslocX;
   143	        int y2 = (int) (p2.y - uy * RAIO) + deslocY;
   144	
   145	        g2.drawLine(x1, y1, x2, y2);
   146	
   147	        if (grafo.isDirigido()) {
   148	            desenharSeta(g2, x2, y2, ux, uy);
   149	        }
   150	
   151	        escrever(g2, texto, (x1 + x2) / 2, (y1 + y2) / 2);
   152	    }

   154	    private void desenharSeta(Graphics2D g2, int x, int y, double ux, double uy) {
   155	        double angulo = Math.atan2(uy, ux);
   156	        double tamanho = 12;
   157	        double abertura = Math.toRadians(22);
   158	
   159	        int[] xs = {x, (int) (x - tamanho * Math.cos(angulo - abertura)),
   160	                (int) (x - tamanho * Math.cos(angulo + abertura))};
   161	        int[] ys = {y, (int) (y - tamanho * Math.sin(angulo - abertura)),
   162	                (int) (y - tamanho * Math.sin(angulo + abertura))};
   163	
   164	        g2.fillPolygon(xs, ys, 3);
   165	    }

   286	    private void busca() {
   287	        String origem = perguntar("Vértice de saída:");
   288	        if (origem == null) {
   289	            return;
   290	        }
   291	        String destino = perguntar("Vértice de chegada:");
   292	        if (destino == null) {
   293	            return;
   294	        }
   295	        try {
   296	            Grafo.ResultadoBusca b = grafo.buscaEmProfundidade(origem, destino);
   297	
   298	            String texto = "===== BUSCA EM PROFUNDIDADE GUIADA =====\n";
   299	            texto += "Saída: " + b.getOrigem().getId()
   300	                    + "   Chegada: " + b.getDestino().getId() + "\n";
   301	            texto += "Ordem de visita: " + b.getOrdemVisita() + "\n";
   302	            texto += "Arestas da árvore:\n";
   303	            for (Aresta a : b.getArestasArvore()) {
   304	                texto += "  " + a.getId() + ": " + a.getOrigem().getId()
   305	                        + (grafo.isDirigido() ? " -> " : " -- ") + a.getDestino().getId()
   306	                        + "  peso " + a.getPesoFormatado() + "\n";
   307	            }
   308	            if (b.isEncontrou()) {
   309	                texto += "Caminho: " + b.getCaminhoVertices()
   310	                        + "   custo " + Aresta.formatarPeso(b.getCustoCaminho());
   311	            } else {
   312	                texto += b.getDestino().getId() + " não é alcançável a partir de "
   313	                        + b.getOrigem().getId() + ".";
   314	            }
   315	            imprimir(texto);
   316	
   317	            Map<String, Color> cores = new HashMap<>();
   318	            cores.put(b.getOrigem().getId(), new Color(0xA8, 0xE0, 0xB0));
   319	            cores.put(b.getDestino().getId(), new Color(0xF5, 0xB4, 0xA8));
   320	            painel.realcar(b.getArestasArvore(), idsDe(b.getCaminhoArestas()), cores);
   321	        } catch (RuntimeException ex) {
   322	            erro(ex);
   323	        }
   324	    }

   326	    private void roy() {
   327	        List<List<Vertice>> componentes = grafo.royComponentes();
   328	
   329	        String texto = "===== ROY: COMPONENTES "
   330	                + (grafo.isDirigido() ? "FORTEMENTE CONEXAS" : "CONEXAS") + " =====\n";
   331	
   332	        Map<String, Color> cores = new HashMap<>();
   333	        for (int i = 0; i < componentes.size(); i++) {
   334	            texto += "C" + (i + 1) + " = " + componentes.get(i) + "\n";
   335	            for (Vertice v : componentes.get(i)) {
   336	                cores.put(v.getId(), PainelGrafo.PALETA[i % PainelGrafo.PALETA.length]);
   337	            }
   338	        }
   339	        texto += "Total de componentes: " + componentes.size();
   340	        imprimir(texto);
   341	
   342	        painel.realcar(null, null, cores);
   343	    }
