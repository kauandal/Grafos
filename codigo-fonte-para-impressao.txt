   426	     * Algoritmo de Prim a partir de um vértice inicial.
   427	     *
   428	     * Prim é definido para grafo não dirigido. Em grafo dirigido o
   429	     * algoritmo é aplicado sobre o grafo subjacente, ignorando o sentido.
   430	     * Se o grafo for desconexo, o resultado é uma floresta geradora
   431	     * mínima, com uma árvore por componente.
   432	     */
   433	    public ResultadoAGM prim(String idInicial) {
   434	        if (vertices.isEmpty()) {
   435	            throw new IllegalStateException("O grafo não possui vértices.");
   436	        }
   437	
   438	        Vertice inicial = (idInicial == null) ? vertices.get(0) : buscarVertice(idInicial);
   439	        if (inicial == null) {
   440	            throw new IllegalArgumentException("Vértice inicial não existe: " + idInicial);
   441	        }
   442	
   443	        Map<Vertice, List<Aresta>> incidencia = mapaIncidencia(false);
   444	        Set<Vertice> visitados = new LinkedHashSet<>();
   445	        List<Aresta> arvore = new ArrayList<>();
   446	        double custo = 0.0;
   447	        int componentes = 0;
   448	
   449	        // O vértice escolhido é a raiz da primeira árvore; os demais
   450	        // servem de raiz apenas se o grafo for desconexo.
   451	        List<Vertice> raizes = new ArrayList<>();
   452	        raizes.add(inicial);
   453	        for (Vertice v : vertices) {
   454	            if (v != inicial) {
   455	                raizes.add(v);
   456	            }
   457	        }
   458	
   459	        for (Vertice raiz : raizes) {
   460	            if (visitados.contains(raiz)) {
   461	                continue;
   462	            }
   463	            componentes++;
   464	
   465	            PriorityQueue<Aresta> fila = new PriorityQueue<>(
   466	                    Comparator.comparingDouble(Aresta::getPeso).thenComparing(Aresta::getId));
   467	
   468	            expandir(raiz, visitados, incidencia, fila);
   469	
   470	            while (!fila.isEmpty()) {
   471	                Aresta a = fila.poll();
   472	
   473	                Vertice fora = null;
   474	                if (!visitados.contains(a.getOrigem())) {
   475	                    fora = a.getOrigem();
   476	                } else if (!visitados.contains(a.getDestino())) {
   477	                    fora = a.getDestino();
   478	                }
   479	
   480	                // Aresta com as duas extremidades já na árvore formaria ciclo.
   481	                if (fora == null) {
   482	                    continue;
   483	                }
   484	
   485	                arvore.add(a);
   486	                custo += a.getPeso();
   487	                expandir(fora, visitados, incidencia, fila);
   488	            }
   489	        }
   490	
   491	        return new ResultadoAGM(inicial, arvore, custo, componentes);
   492	    }

   494	    /** Marca o vértice como pertencente à árvore e enfileira suas arestas candidatas. */
   495	    private void expandir(Vertice v, Set<Vertice> visitados,
   496	                          Map<Vertice, List<Aresta>> incidencia,
   497	                          PriorityQueue<Aresta> fila) {
   498	        visitados.add(v);
   499	        for (Aresta a : incidencia.get(v)) {
   500	            if (!visitados.contains(a.oposto(v))) {
   501	                fila.add(a);
   502	            }
   503	        }
   504	    }

   510	    /**
   511	     * Busca em profundidade com vértice de saída e de chegada.
   512	     * A busca é interrompida assim que o vértice de chegada é alcançado.
   513	     * Em grafo dirigido o sentido dos arcos é respeitado.
   514	     */
   515	    public ResultadoBusca buscaEmProfundidade(String idOrigem, String idDestino) {
   516	        Vertice origem = buscarVertice(idOrigem);
   517	        Vertice destino = buscarVertice(idDestino);
   518	
   519	        if (origem == null) {
   520	            throw new IllegalArgumentException("Vértice de saída não existe: " + idOrigem);
   521	        }
   522	        if (destino == null) {
   523	            throw new IllegalArgumentException("Vértice de chegada não existe: " + idDestino);
   524	        }
   525	
   526	        Map<Vertice, List<Aresta>> incidencia = mapaIncidencia(dirigido);
   527	
   528	        Set<Vertice> visitados = new LinkedHashSet<>();
   529	        List<Vertice> ordem = new ArrayList<>();
   530	        List<Aresta> arvore = new ArrayList<>();
   531	        Map<Vertice, Aresta> chegada = new HashMap<>();
   532	
   533	        boolean encontrou = explorar(origem, destino, incidencia, visitados, ordem, arvore, chegada);
   534	
   535	        List<Aresta> caminhoArestas = new ArrayList<>();
   536	        List<Vertice> caminhoVertices = new ArrayList<>();
   537	        double custoCaminho = 0.0;
   538	
   539	        if (encontrou) {
   540	            Vertice atual = destino;
   541	            caminhoVertices.add(atual);
   542	            while (chegada.containsKey(atual)) {
   543	                Aresta a = chegada.get(atual);
   544	                caminhoArestas.add(0, a);
   545	                custoCaminho += a.getPeso();
   546	                atual = a.oposto(atual);
   547	                caminhoVertices.add(0, atual);
   548	            }
   549	        }
   550	
   551	        return new ResultadoBusca(origem, destino, encontrou, ordem, arvore,
   552	                caminhoVertices, caminhoArestas, custoCaminho);
   553	    }

   555	    /**
   556	     * Percurso em profundidade com pilha explícita, o que evita estouro de
   557	     * pilha de recursão em grafos grandes. A ordem de visita é a mesma da
   558	     * versão recursiva: segue sempre pela primeira aresta ainda não explorada.
   559	     */
   560	    private boolean explorar(Vertice origem, Vertice destino,
   561	                             Map<Vertice, List<Aresta>> incidencia,
   562	                             Set<Vertice> visitados, List<Vertice> ordem,
   563	                             List<Aresta> arvore, Map<Vertice, Aresta> chegada) {
   564	
   565	        visitados.add(origem);
   566	        ordem.add(origem);
   567	
   568	        if (origem == destino) {
   569	            return true;
   570	        }
   571	
   572	        Deque<Vertice> pilhaVertices = new ArrayDeque<>();
   573	        Deque<Iterator<Aresta>> pilhaArestas = new ArrayDeque<>();
   574	
   575	        pilhaVertices.push(origem);
   576	        pilhaArestas.push(incidencia.get(origem).iterator());
   577	
   578	        while (!pilhaVertices.isEmpty()) {
   579	            Vertice atual = pilhaVertices.peek();
   580	            Iterator<Aresta> it = pilhaArestas.peek();
   581	            boolean avancou = false;
   582	
   583	            while (it.hasNext()) {
   584	                Aresta a = it.next();
   585	                Vertice proximo = a.oposto(atual);
   586	
   587	                if (visitados.contains(proximo)) {
   588	                    continue;
   589	                }
   590	
   591	                visitados.add(proximo);
   592	                ordem.add(proximo);
   593	                arvore.add(a);
   594	                chegada.put(proximo, a);
   595	
   596	                if (proximo == destino) {
   597	                    return true;
   598	                }
   599	
   600	                pilhaVertices.push(proximo);
   601	                pilhaArestas.push(incidencia.get(proximo).iterator());
   602	                avancou = true;
   603	                break;
   604	            }
   605	
   606	            if (!avancou) {
   607	                pilhaVertices.pop();
   608	                pilhaArestas.pop();
   609	            }
   610	        }
   611	
   612	        return false;
   613	    }

   619	    /**
   620	     * Matriz de alcançabilidade (fecho transitivo) pelo algoritmo de Roy,
   621	     * também conhecido como Roy-Warshall, aplicado sobre a matriz booleana
   622	     * de adjacência.
   623	     */
   624	    public boolean[][] matrizAlcancabilidade() {
   625	        int n = vertices.size();
   626	        boolean[][] r = new boolean[n][n];
   627	        Map<Vertice, Integer> indices = indices();
   628	
   629	        // Todo vértice alcança a si mesmo.
   630	        for (int i = 0; i < n; i++) {
   631	            r[i][i] = true;
   632	        }
   633	
   634	        for (Aresta a : arestas) {
   635	            int i = indices.get(a.getOrigem());
   636	            int j = indices.get(a.getDestino());
   637	            r[i][j] = true;
   638	            if (!dirigido) {
   639	                r[j][i] = true;
   640	            }
   641	        }
   642	
   643	        // Fecho transitivo: se i alcança k e k alcança j, então i alcança j.
   644	        for (int k = 0; k < n; k++) {
   645	            for (int i = 0; i < n; i++) {
   646	                if (!r[i][k]) {
   647	                    continue;
   648	                }
   649	                for (int j = 0; j < n; j++) {
   650	                    if (r[k][j]) {
   651	                        r[i][j] = true;
   652	                    }
   653	                }
   654	            }
   655	        }
   656	        return r;
   657	    }

   659	    /**
   660	     * Componentes conexas (grafo não dirigido) ou fortemente conexas
   661	     * (grafo dirigido): v e w pertencem ao mesmo conjunto quando
   662	     * v alcança w e w alcança v na matriz de alcançabilidade.
   663	     */
   664	    public List<List<Vertice>> royComponentes() {
   665	        boolean[][] r = matrizAlcancabilidade();
   666	        int n = vertices.size();
   667	        boolean[] agrupado = new boolean[n];
   668	        List<List<Vertice>> componentes = new ArrayList<>();
   669	
   670	        for (int i = 0; i < n; i++) {
   671	            if (agrupado[i]) {
   672	                continue;
   673	            }
   674	            List<Vertice> componente = new ArrayList<>();
   675	            for (int j = 0; j < n; j++) {
   676	                if (!agrupado[j] && r[i][j] && r[j][i]) {
   677	                    agrupado[j] = true;
   678	                    componente.add(vertices.get(j));
   679	                }
   680	            }
   681	            componentes.add(componente);
   682	        }
   683	        return componentes;
   684	    }

   686	    public String formatarMatrizAlcancabilidade() {
   687	        if (vertices.isEmpty()) {
   688	            return "O grafo não possui vértices.";
   689	        }
   690	
   691	        boolean[][] r = matrizAlcancabilidade();
   692	        StringBuilder sb = new StringBuilder();
   693	
   694	        sb.append("===== MATRIZ DE ALCANÇABILIDADE (ROY) =====\n");
   695	        sb.append(String.format("%8s", ""));
   696	        for (Vertice v : vertices) {
   697	            sb.append(String.format("%10s", v.getId()));
   698	        }
   699	        sb.append('\n');
   700	
   701	        for (int i = 0; i < r.length; i++) {

   703	            for (int j = 0; j < r[i].length; j++) {

   705	            }
   706	            sb.append('\n');

   220	            g2.drawString("Grafo sem vértices. Use o menu Vértice para inserir.", 24, 34);
   221	            g2.dispose();
   222	            return;
   223	        }
   224	
   225	        garantirPosicoes();
   226	
   227	        List<Aresta> lista = (arestasVisiveis != null) ? arestasVisiveis : grafo.getArestas();
   228	
   229	        // Arestas entre o mesmo par são curvadas em leque para não se sobrepor.
   230	        Map<String, List<Aresta>> grupos = new LinkedHashMap<>();
   231	        for (Aresta a : lista) {
   232	            grupos.computeIfAbsent(chavePar(a), k -> new ArrayList<>()).add(a);
   233	        }
   234	
   235	        for (List<Aresta> grupo : grupos.values()) {
   236	            int total = grupo.size();
   237	            for (int i = 0; i < total; i++) {
   238	                Aresta a = grupo.get(i);
   239	                double deslocamento = (i - (total - 1) / 2.0) * 64.0;
   240	
   241	                // A perpendicular usada na curvatura segue o sentido origem -> destino.
   242	                // Arestas do mesmo par em sentido invertido precisam do deslocamento
   243	                // espelhado, senão as duas curvas caem uma sobre a outra.
   244	                if (a.getOrigem().getId().compareTo(a.getDestino().getId()) > 0) {
   245	                    deslocamento = -deslocamento;
   246	                }
   247	
   248	                desenharAresta(g2, a, deslocamento);
   249	            }
   250	        }
   251	
   252	        for (Vertice v : grafo.getVertices()) {
   253	            desenharVertice(g2, v);
   254	        }
   255	
   256	        g2.dispose();
   257	    }

   259	    private String chavePar(Aresta a) {
   260	        String x = a.getOrigem().getId();
   261	        String y = a.getDestino().getId();
   262	        String menor = (x.compareTo(y) <= 0) ? x : y;
   263	        String maior = (x.compareTo(y) <= 0) ? y : x;
   264	        return menor.length() + ":" + menor + ":" + maior;
   265	    }

   267	    private void desenharAresta(Graphics2D g2, Aresta a, double deslocamento) {
   268	        Point2D.Double p1 = posicoes.get(a.getOrigem().getId());
   269	        Point2D.Double p2 = posicoes.get(a.getDestino().getId());
   270	
   271	        if (p1 == null || p2 == null) {
   272	            return;
   273	        }
   274	
   275	        boolean destaque = arestasDestacadas.contains(a.getId());
   276	        g2.setColor(destaque ? COR_DESTAQUE : COR_ARESTA);
   277	        g2.setStroke(new BasicStroke(destaque ? 3.2f : 1.6f));
   278	
   279	        if (a.isLaco()) {
   280	            desenharLaco(g2, a, p1, destaque);
   281	            return;
   282	        }
   283	
   284	        double dx = p2.x - p1.x;
   285	        double dy = p2.y - p1.y;
   286	        double comprimento = Math.hypot(dx, dy);
   287	        if (comprimento < 1e-6) {
   288	            return;
   289	        }
   290	
   291	        // Ponto de controle deslocado na perpendicular gera a curvatura.
   292	        double cx = (p1.x + p2.x) / 2.0 - (dy / comprimento) * deslocamento;
   293	        double cy = (p1.y + p2.y) / 2.0 + (dx / comprimento) * deslocamento;
   294	
   295	        List<Point2D.Double> pontos = amostrar(new QuadCurve2D.Double(p1.x, p1.y, cx, cy, p2.x, p2.y), 48);
   296	
   297	        int inicio = 0;
   298	        while (inicio < pontos.size() - 1 && pontos.get(inicio).distance(p1) < RAIO) {
   299	            inicio++;
   300	        }
   301	        int fim = pontos.size() - 1;
   302	        while (fim > inicio && pontos.get(fim).distance(p2) < RAIO) {
   303	            fim--;
   304	        }
   305	
   306	        Path2D.Double caminho = new Path2D.Double();
   307	        caminho.moveTo(pontos.get(inicio).x, pontos.get(inicio).y);
   308	        for (int k = inicio + 1; k <= fim; k++) {
   309	            caminho.lineTo(pontos.get(k).x, pontos.get(k).y);
   310	        }
   311	        g2.draw(caminho);
   312	
   313	        if (grafo.isDirigido() && fim > inicio) {
   314	            desenharSeta(g2, pontos.get(fim - 1), pontos.get(fim));
   315	        }
   316	
   317	        Point2D.Double meio = pontos.get((inicio + fim) / 2);
   318	        desenharRotulo(g2, rotulo(a), meio.x, meio.y, destaque);
   319	    }

   321	    private void desenharLaco(Graphics2D g2, Aresta a, Point2D.Double p, boolean destaque) {
   322	        double largura = 40;
   323	        double altura = 34;
   324	        double topo = p.y - RAIO - altura + 8;
   325	
   326	        g2.draw(new Ellipse2D.Double(p.x - largura / 2, topo, largura, altura));
   327	
   328	        if (grafo.isDirigido()) {
   329	            Point2D.Double de = new Point2D.Double(p.x + largura / 2 - 6, topo + altura - 10);
   330	            Point2D.Double ate = new Point2D.Double(p.x + 6, p.y - RAIO + 2);
   331	            desenharSeta(g2, de, ate);
   332	        }
   333	
   334	        desenharRotulo(g2, rotulo(a), p.x, topo - 8, destaque);
   335	    }

   337	    private void desenharSeta(Graphics2D g2, Point2D.Double de, Point2D.Double ate) {
   338	        double angulo = Math.atan2(ate.y - de.y, ate.x - de.x);
   339	        double tamanho = 13;
   340	        double abertura = Math.toRadians(22);
   341	
   342	        Path2D.Double seta = new Path2D.Double();
   343	        seta.moveTo(ate.x, ate.y);
   344	        seta.lineTo(ate.x - tamanho * Math.cos(angulo - abertura), ate.y - tamanho * Math.sin(angulo - abertura));
   345	        seta.lineTo(ate.x - tamanho * Math.cos(angulo + abertura), ate.y - tamanho * Math.sin(angulo + abertura));
   346	        seta.closePath();
   347	        g2.fill(seta);
   348	    }

   351	        Font fonteAnterior = g2.getFont();
   352	        Color corAnterior = g2.getColor();
   353

   355	        FontMetrics fm = g2.getFontMetrics();
   356	        int largura = fm.stringWidth(texto);
   357	        int altura = fm.getAscent();
   358

   361

   363	        g2.drawString(texto, (int) x - largura / 2, (int) y + altura / 2 - 1);

   365	        g2.setFont(fonteAnterior);
   366	        g2.setColor(corAnterior);
   367	    }

   370	        Point2D.Double p = posicoes.get(v.getId());
   371	        if (p == null) {
   372	            return;
   373	        }
   374	
   375	        Color preenchimento = coresVertices.getOrDefault(v.getId(), COR_VERTICE);
   376	        Ellipse2D.Double circulo = new Ellipse2D.Double(p.x - RAIO, p.y - RAIO, RAIO * 2, RAIO * 2);

   395	        double luminancia = 0.299 * fundo.getRed() + 0.587 * fundo.getGreen() + 0.114 * fundo.getBlue();
   396	        return (luminancia < 140) ? Color.WHITE : new Color(0x1B, 0x2A, 0x38);

   403	    /** Converte a curva em uma sequência de pontos, o que facilita recortar e apontar a seta. */
   404	    private List<Point2D.Double> amostrar(QuadCurve2D.Double curva, int divisoes) {
   405	        List<Point2D.Double> pontos = new ArrayList<>(divisoes + 1);
   406	        for (int i = 0; i <= divisoes; i++) {
   407	            double t = i / (double) divisoes;
   408	            double u = 1 - t;
   409	            double x = u * u * curva.x1 + 2 * u * t * curva.ctrlx + t * t * curva.x2;
   410	            double y = u * u * curva.y1 + 2 * u * t * curva.ctrly + t * t * curva.y2;
   411	            pontos.add(new Point2D.Double(x, y));
   412	        }
   413	        return pontos;
   414	    }

   327	    private void executarPrim() {
   328	        if (grafo.getVertices().isEmpty()) {
   329	            avisar("O grafo não possui vértices.");
   330	            return;
   331	        }
   332	
   333	        if (grafo.isDirigido()) {
   334	            int opcao = JOptionPane.showConfirmDialog(this,
   335	                    "O algoritmo de Prim é definido para grafo não dirigido.\n"
   336	                            + "Aplicar sobre o grafo subjacente, ignorando o sentido dos arcos?",
   337	                    "Grafo dirigido", JOptionPane.YES_NO_OPTION, JOptionPane.WARNING_MESSAGE);
   338	            if (opcao != JOptionPane.YES_OPTION) {
   339	                return;
   340	            }
   341	        }
   342	
   343	        String inicial = escolherVertice("Algoritmo de Prim", "Vértice inicial:");
   344	        if (inicial == null) {
   345	            return;
   346	        }
   347	
   348	        try {
   349	            Grafo.ResultadoAGM resultado = grafo.prim(inicial);
   350	
   351	            StringBuilder sb = new StringBuilder();
   352	            sb.append("===== ALGORITMO DE PRIM: ÁRVORE GERADORA MÍNIMA =====\n");
   353	            sb.append("Vértice inicial: ").append(resultado.getInicial().getId()).append('\n');
   354	
   355	            if (resultado.getArestas().isEmpty()) {
   356	                sb.append("Nenhuma aresta selecionada: não há ligações entre os vértices.\n");
   357	            } else {
   358	                sb.append("Arestas da AGM:\n");
   359	                for (Aresta a : resultado.getArestas()) {
   360	                    sb.append("  ").append(a.getId())
   361	                            .append(": ").append(a.getOrigem().getId())
   362	                            .append(" -- ").append(a.getDestino().getId())
   363	                            .append("  peso ").append(a.getPesoFormatado()).append('\n');
   364	                }
   365	            }
   366	
   367	            sb.append("Total de arestas: ").append(resultado.getArestas().size()).append('\n');
   368	            sb.append("CUSTO TOTAL DA AGM: ").append(Aresta.formatarPeso(resultado.getCusto()));
   369	
   370	            if (resultado.isFloresta()) {
   371	                sb.append("\nAtenção: o grafo é desconexo (").append(resultado.getComponentes())
   372	                        .append(" componentes). O resultado é uma floresta geradora mínima,\n")
   373	                        .append("com uma árvore mínima por componente.");
   374	            }
   375	
   376	            imprimir(sb.toString());
   377	
   378	            Set<String> ids = new LinkedHashSet<>();
   379	            for (Aresta a : resultado.getArestas()) {
   380	                ids.add(a.getId());
   381	            }
   382	
   383	            String rodape = "Custo total da AGM: " + Aresta.formatarPeso(resultado.getCusto())
   384	                    + "   |   Arestas: " + resultado.getArestas().size()
   385	                    + "   |   Vértice inicial: " + resultado.getInicial().getId()
   386	                    + (resultado.isFloresta()
   387	                    ? "   |   Floresta com " + resultado.getComponentes() + " árvores" : "");
   388	
   389	            abrirJanelaResultado("Árvore Geradora Mínima (Prim)",
   390	                    resultado.getArestas(), ids, null, rodape);
   391	
   392	        } catch (RuntimeException ex) {
   393	            erro(ex);
   394	        }
   395	    }

   401	    private void executarBusca() {
   402	        if (grafo.getVertices().isEmpty()) {
   403	            avisar("O grafo não possui vértices.");
   404	            return;
   405	        }
   406	
   407	        String origem = escolherVertice("Busca em profundidade", "Vértice de saída:");
   408	        if (origem == null) {
   409	            return;
   410	        }
   411	        String destino = escolherVertice("Busca em profundidade", "Vértice de chegada:");
   412	        if (destino == null) {
   413	            return;
   414	        }
   415	
   416	        try {
   417	            Grafo.ResultadoBusca resultado = grafo.buscaEmProfundidade(origem, destino);
   418	
   419	            StringBuilder sb = new StringBuilder();
   420	            sb.append("===== BUSCA EM PROFUNDIDADE GUIADA =====\n");
   421	            sb.append("Vértice de saída: ").append(resultado.getOrigem().getId()).append('\n');
   422	            sb.append("Vértice de chegada: ").append(resultado.getDestino().getId()).append('\n');
   423	
   424	            sb.append("Ordem de visita: ");
   425	            for (int i = 0; i < resultado.getOrdemVisita().size(); i++) {
   426	                sb.append(i > 0 ? " > " : "").append(resultado.getOrdemVisita().get(i).getId());
   427	            }
   428	            sb.append('\n');
   429	
   430	            sb.append("Arestas da árvore de busca:\n");
   431	            if (resultado.getArestasArvore().isEmpty()) {
   432	                sb.append("  (nenhuma)\n");
   433	            } else {
   434	                for (Aresta a : resultado.getArestasArvore()) {
   435	                    sb.append("  ").append(a.getId())
   436	                            .append(": ").append(a.getOrigem().getId())
   437	                            .append(grafo.isDirigido() ? " -> " : " -- ")
   438	                            .append(a.getDestino().getId())
   439	                            .append("  peso ").append(a.getPesoFormatado()).append('\n');
   440	                }
   441	            }
   442	
   443	            String rodape;
   444	            if (resultado.isEncontrou()) {
   445	                StringBuilder caminho = new StringBuilder();
   446	                for (int i = 0; i < resultado.getCaminhoVertices().size(); i++) {
   447	                    caminho.append(i > 0 ? " -> " : "").append(resultado.getCaminhoVertices().get(i).getId());
   448	                }
   449	                sb.append("CAMINHO ENCONTRADO: ").append(caminho).append('\n');
   450	                sb.append("Custo do caminho: ").append(Aresta.formatarPeso(resultado.getCustoCaminho()));
   451	
   452	                rodape = "Caminho: " + caminho
   453	                        + "   |   Custo: " + Aresta.formatarPeso(resultado.getCustoCaminho())
   454	                        + "   |   Vértices visitados: " + resultado.getOrdemVisita().size();
   455	            } else {
   456	                sb.append("CAMINHO NÃO ENCONTRADO: ")
   457	                        .append(resultado.getDestino().getId())
   458	                        .append(" não é alcançável a partir de ")
   459	                        .append(resultado.getOrigem().getId()).append('.');
   460	
   461	                rodape = "Vértice de chegada não alcançável a partir de "
   462	                        + resultado.getOrigem().getId()
   463	                        + "   |   Vértices visitados: " + resultado.getOrdemVisita().size();
   464	            }
   465	
   466	            imprimir(sb.toString());
   467	
   468	            Set<String> destaques = new LinkedHashSet<>();
   469	            for (Aresta a : resultado.getCaminhoArestas()) {
   470	                destaques.add(a.getId());
   471	            }
   472	
   473	            Map<String, Color> cores = new HashMap<>();
   474	            cores.put(resultado.getOrigem().getId(), new Color(0xA8, 0xE0, 0xB0));
   475	            cores.put(resultado.getDestino().getId(), new Color(0xF5, 0xB4, 0xA8));
   476	
   477	            abrirJanelaResultado("Árvore da busca em profundidade guiada",
   478	                    resultado.getArestasArvore(), destaques, cores, rodape);
   479	
   480	        } catch (RuntimeException ex) {
   481	            erro(ex);
   482	        }
   483	    }

   489	    private void executarRoy() {
   490	        if (grafo.getVertices().isEmpty()) {
   491	            avisar("O grafo não possui vértices.");
   492	            return;
   493	        }
   494	
   495	        List<List<Vertice>> componentes = grafo.royComponentes();
   496	
   497	        String tipo = grafo.isDirigido()
   498	                ? "COMPONENTES FORTEMENTE CONEXAS"
   499	                : "COMPONENTES CONEXAS";
   500	
   501	        StringBuilder sb = new StringBuilder();
   502	        sb.append("===== ALGORITMO DE ROY: ").append(tipo).append(" =====\n");
   503	        sb.append(grafo.formatarMatrizAlcancabilidade()).append('\n');
   504	        sb.append("\nTotal de componentes: ").append(componentes.size()).append('\n');
   505	
   506	        Map<String, Color> cores = new HashMap<>();
   507	        for (int i = 0; i < componentes.size(); i++) {
   508	            Color cor = PainelGrafo.PALETA_COMPONENTES[i % PainelGrafo.PALETA_COMPONENTES.length];
   509	
   510	            sb.append("C").append(i + 1).append(" = { ");
   511	            List<Vertice> componente = componentes.get(i);
   512	            for (int j = 0; j < componente.size(); j++) {
   513	                sb.append(j > 0 ? ", " : "").append(componente.get(j).getId());
   514	                cores.put(componente.get(j).getId(), cor);
   515	            }
   516	            sb.append(" }\n");
   517	        }
   518	
   519	        if (componentes.size() == 1) {
   520	            sb.append(grafo.isDirigido()
   521	                    ? "O grafo é fortemente conexo."
   522	                    : "O grafo é conexo.");
   523	        } else {
   524	            sb.append(grafo.isDirigido()
   525	                    ? "O grafo não é fortemente conexo."
   526	                    : "O grafo é desconexo.");
   527	        }
   528	
   529	        imprimir(sb.toString());
   530	
   531	        painelGrafo.setCoresVertices(cores);
   532	
   533	        String rodape = tipo.charAt(0) + tipo.substring(1).toLowerCase()
   534	                + ": " + componentes.size()
   535	                + "   |   cada cor representa um conjunto";
   536	
   537	        abrirJanelaResultado("Roy: " + tipo.toLowerCase(),
   538	                new ArrayList<>(grafo.getArestas()), null, cores, rodape);
   539	    }

   545	    private void abrirJanelaResultado(String titulo, List<Aresta> arestas,

   555	                Math.max(painelGrafo.getWidth(), 780),
   556	                Math.max(painelGrafo.getHeight(), 520)));
   557	
   558	        JPanel conteudo = new JPanel(new BorderLayout(8, 8));

   566	        info.setFont(info.getFont().deriveFont(Font.BOLD, 13f));

   572	        conteudo.add(info, BorderLayout.SOUTH);

   575	        janela.setContentPane(conteudo);
   576	        janela.pack();
   577	        janela.setLocationRelativeTo(this);
   578	        janela.setVisible(true);
   579	    }

   614	    private String escolherVertice(String titulo, String mensagem) {

   629	        List<Aresta> lista = grafo.getArestas();
