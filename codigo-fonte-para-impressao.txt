    76	    /**
    77	     Distância em linha reta, em quilômetros, entre este vértice e outro,
    78	     pela fórmula de Haversine.
    79	
    80	     Haversine mede sobre a superfície da esfera, então o valor nunca é maior
    81	     que a distância percorrida por estrada. É isso que torna a heurística
    82	     admissível e garante que o A* encontre o caminho de menor custo.
    83	
    84	     Quando algum dos dois vértices não tem coordenada, devolve zero: sem
    85	     heurística o A* se comporta como o algoritmo de Dijkstra, que continua
    86	     correto, apenas explora mais vértices.
    87	     */
    88	    public double distanciaHaversine(Vertice outro) {
    89	        if (!this.temCoordenadas || !outro.temCoordenadas) {
    90	            return 0.0;
    91	        }
    92	
    93	        double deltaLat = Math.toRadians(outro.latitude - this.latitude);
    94	        double deltaLon = Math.toRadians(outro.longitude - this.longitude);
    95	
    96	        double a = Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2)
    97	                + Math.cos(Math.toRadians(this.latitude))
    98	                * Math.cos(Math.toRadians(outro.latitude))
    99	                * Math.sin(deltaLon / 2) * Math.sin(deltaLon / 2);
   100	
   101	        double c = 2 * Math.asin(Math.min(1.0, Math.sqrt(a)));
   102	
   103	        return RAIO_TERRA_KM * c;
   104	    }

   108	        return id;
   109	    }

   272	    /**
   273	     * Algoritmo de Prim sobre o grafo.
   274	     *
   275	     * A árvore geradora mínima é propriedade do grafo: o custo total é o
   276	     * mesmo independente do vértice por onde a construção começa, por isso
   277	     * o ponto de partida não é parâmetro. A árvore cresce a partir do
   278	     * primeiro vértice da lista.
   279	     *
   280	     * Prim é definido para grafo não dirigido. Em grafo dirigido o
   281	     * algoritmo é aplicado sobre o grafo subjacente, ignorando o sentido.
   282	     * Se o grafo for desconexo, o resultado é uma floresta geradora
   283	     * mínima, com uma árvore por componente.
   284	     */
   285	    public ResultadoAGM prim() {
   286	        if (vertices.isEmpty()) {
   287	            throw new IllegalStateException("O grafo não possui vértices.");
   288	        }
   289	
   290	        Map<Vertice, List<Aresta>> incidencia = incidenciaSemSentido();
   291	        Set<Vertice> visitados = new HashSet<>();
   292	        List<Aresta> arvore = new ArrayList<>();
   293	        double custo = 0.0;
   294	        int componentes = 0;
   295	
   296	        // O primeiro vértice é a raiz da árvore. Os demais só viram raiz de
   297	        // uma nova árvore se o grafo for desconexo e ainda houver vértice
   298	        // fora de tudo o que já foi alcançado.
   299	        for (Vertice raiz : vertices) {
   300	            if (visitados.contains(raiz)) {
   301	                continue;
   302	            }
   303	            componentes++;
   304	
   305	            PriorityQueue<Aresta> fila = new PriorityQueue<>(
   306	                    Comparator.comparingDouble(Aresta::getPeso).thenComparing(Aresta::getId));
   307	
   308	            expandir(raiz, visitados, incidencia, fila);
   309	
   310	            while (!fila.isEmpty()) {
   311	                Aresta a = fila.poll();
   312	
   313	                Vertice fora = null;
   314	                if (!visitados.contains(a.getOrigem())) {
   315	                    fora = a.getOrigem();
   316	                } else if (!visitados.contains(a.getDestino())) {
   317	                    fora = a.getDestino();
   318	                }
   319	
   320	                // Aresta com as duas extremidades já na árvore formaria ciclo.
   321	                if (fora == null) {
   322	                    continue;
   323	                }
   324	
   325	                arvore.add(a);
   326	                custo += a.getPeso();
   327	                expandir(fora, visitados, incidencia, fila);
   328	            }
   329	        }
   330	
   331	        return new ResultadoAGM(arvore, custo, componentes);
   332	    }

   334	    /** Marca o vértice como pertencente à árvore e enfileira suas arestas candidatas. */
   335	    private void expandir(Vertice v, Set<Vertice> visitados,
   336	                          Map<Vertice, List<Aresta>> incidencia,
   337	                          PriorityQueue<Aresta> fila) {
   338	        visitados.add(v);
   339	        for (Aresta a : incidencia.get(v)) {
   340	            if (!visitados.contains(a.oposto(v))) {
   341	                fila.add(a);
   342	            }
   343	        }

   358

   360	            throw new IllegalArgumentException("Vértice de saída não existe: " + idOrigem);
   361	        }

   363	            throw new IllegalArgumentException("Vértice de chegada não existe: " + idDestino);
   364	        }
   365	
   366	        Set<Vertice> visitados = new HashSet<>();
   367	        List<Vertice> ordem = new ArrayList<>();
   368	        List<Aresta> arvore = new ArrayList<>();
   369	        Map<Vertice, Aresta> chegada = new HashMap<>();
   370	
   371	        boolean encontrou = explorar(origem, destino, visitados, ordem, arvore, chegada);
   372	
   373	        List<Aresta> caminhoArestas = new ArrayList<>();
   374	        List<Vertice> caminhoVertices = new ArrayList<>();
   375	        double custoCaminho = 0.0;
   376	
   377	        if (encontrou) {
   378	            Vertice atual = destino;
   379	            caminhoVertices.add(atual);
   380	            while (chegada.containsKey(atual)) {
   381	                Aresta a = chegada.get(atual);
   382	                caminhoArestas.add(0, a);
   383	                custoCaminho += a.getPeso();
   384	                atual = a.oposto(atual);
   385	                caminhoVertices.add(0, atual);
   386	            }
   387	        }
   388	
   389	        return new ResultadoBusca(origem, destino, encontrou, ordem, arvore,
   390	                caminhoVertices, caminhoArestas, custoCaminho);
   391	    }

   393	    /**
   394	     * Percurso recursivo em profundidade sobre a lista de adjacência.
   395	     * Visita o vértice atual, e para cada aresta ainda não explorada desce
   396	     * no vizinho não visitado. Retorna verdadeiro assim que alcança o
   397	     * destino, o que interrompe a busca.
   398	     *
   399	     * Em grafo dirigido a lista de adjacência já contém apenas os arcos que
   400	     * saem do vértice, então o sentido é respeitado sem tratamento extra.
   401	     */
   402	    private boolean explorar(Vertice atual, Vertice destino,
   403	                             Set<Vertice> visitados, List<Vertice> ordem,
   404	                             List<Aresta> arvore, Map<Vertice, Aresta> chegada) {
   405	
   406	        visitados.add(atual);
   407	        ordem.add(atual);
   408	
   409	        if (atual == destino) {
   410	            return true;
   411	        }
   412	
   413	        for (Aresta a : atual.getAdjacencias()) {
   414	            Vertice proximo = a.oposto(atual);
   415	
   416	            if (visitados.contains(proximo)) {
   417	                continue;
   418	            }
   419	
   420	            // Aresta da árvore: foi por ela que o vizinho foi descoberto.
   421	            arvore.add(a);
   422	            chegada.put(proximo, a);
   423	
   424	            if (explorar(proximo, destino, visitados, ordem, arvore, chegada)) {
   425	                return true;
   426	            }
   427	        }
   428	
   429	        return false;
   430	    }

   433	    // ALGORITMO DSATUR: COLORAÇÃO DE VÉRTICES
   434	    // ============================================================
   435	
   436	    /**
   437	     Algoritmo DSATUR de coloração de vértices.
   438	
   439	     Colore os vértices de forma que nenhuma aresta ligue dois vértices da
   440	     mesma cor, tentando usar o menor número possível de cores. A cada passo:
   441	
   442	     1. escolhe o vértice ainda não colorido com maior grau de saturação, que
   443	        é o número de cores distintas já presentes entre os seus vizinhos;
   444	     2. desempata pelo maior grau dentro do subgrafo ainda não colorido;
   445	     3. atribui a menor cor que nenhum vizinho esteja usando.
   446	
   447	     DSATUR é heurística: não garante o número cromático mínimo em todo grafo,
   448	     mas é exato para grafos bipartidos e costuma acertar em grafos pequenos.
   449	
   450	     A coloração ignora o sentido dos arcos, porque dois vértices ligados
   451	     precisam de cores diferentes independentemente da direção. Laços são
   452	     ignorados, já que um vértice não pode ter cor diferente de si mesmo.
   453	     */
   454	    public ResultadoColoracao dsatur() {
   455	        if (vertices.isEmpty()) {
   456	            throw new IllegalStateException("O grafo não possui vértices.");
   457	        }
   458	
   459	        Map<Vertice, List<Aresta>> incidencia = incidenciaSemSentido();
   460	        Map<Vertice, Integer> cores = new LinkedHashMap<>();
   461	
   462	        while (cores.size() < vertices.size()) {
   463	            Vertice escolhido = maisSaturado(incidencia, cores);
   464	            cores.put(escolhido, menorCorLivre(escolhido, incidencia, cores));
   465	        }
   466	
   467	        int total = 0;
   468	        for (int cor : cores.values()) {
   469	            total = Math.max(total, cor + 1);
   470	        }
   471	
   472	        return new ResultadoColoracao(cores, total);
   473	    }

   475	    /** Vértice não colorido com maior saturação, desempatando pelo maior grau. */
   476	    private Vertice maisSaturado(Map<Vertice, List<Aresta>> incidencia, Map<Vertice, Integer> cores) {
   477	        Vertice melhor = null;
   478	        int melhorSaturacao = -1;
   479	        int melhorGrau = -1;
   480	
   481	        for (Vertice v : vertices) {
   482	            if (cores.containsKey(v)) {
   483	                continue;
   484	            }
   485	
   486	            int saturacao = saturacao(v, incidencia, cores);
   487	            int grau = grauEntreNaoColoridos(v, incidencia, cores);
   488	
   489	            if (saturacao > melhorSaturacao
   490	                    || (saturacao == melhorSaturacao && grau > melhorGrau)) {
   491	                melhor = v;
   492	                melhorSaturacao = saturacao;
   493	                melhorGrau = grau;
   494	            }
   495	        }
   496	        return melhor;
   497	    }

   499	    /** Grau de saturação: quantas cores distintas já aparecem entre os vizinhos. */
   500	    private int saturacao(Vertice v, Map<Vertice, List<Aresta>> incidencia, Map<Vertice, Integer> cores) {
   501	        List<Integer> distintas = new ArrayList<>();
   502	
   503	        for (Aresta a : incidencia.get(v)) {
   504	            Vertice vizinho = a.oposto(v);
   505	            if (vizinho == v || !cores.containsKey(vizinho)) {
   506	                continue;
   507	            }
   508	            if (!distintas.contains(cores.get(vizinho))) {
   509	                distintas.add(cores.get(vizinho));
   510	            }
   511	        }
   512	        return distintas.size();
   513	    }

   515	    /** Grau do vértice contando apenas os vizinhos ainda sem cor. */
   516	    private int grauEntreNaoColoridos(Vertice v, Map<Vertice, List<Aresta>> incidencia,
   517	                                      Map<Vertice, Integer> cores) {
   518	        int grau = 0;
   519	        for (Aresta a : incidencia.get(v)) {
   520	            Vertice vizinho = a.oposto(v);
   521	            if (vizinho != v && !cores.containsKey(vizinho)) {
   522	                grau++;
   523	            }
   524	        }
   525	        return grau;
   526	    }

   528	    /** Menor número de cor que nenhum vizinho já colorido esteja usando. */
   529	    private int menorCorLivre(Vertice v, Map<Vertice, List<Aresta>> incidencia,
   530	                              Map<Vertice, Integer> cores) {
   531	        List<Integer> usadas = new ArrayList<>();
   532	
   533	        for (Aresta a : incidencia.get(v)) {
   534	            Vertice vizinho = a.oposto(v);
   535	            if (vizinho != v && cores.containsKey(vizinho)) {
   536	                usadas.add(cores.get(vizinho));
   537	            }
   538	        }
   539	
   540	        int cor = 0;
   541	        while (usadas.contains(cor)) {
   542	            cor++;
   543	        }
   544	        return cor;
   545	    }

   548	     Confere se a coloração é própria, isto é, se nenhuma aresta liga dois
   549	     vértices da mesma cor. Laços são desconsiderados.
   550	     */
   551	    public boolean coloracaoValida(Map<Vertice, Integer> cores) {
   552	        for (Aresta a : arestas) {
   553	            if (a.isLaco()) {
   554	                continue;
   555	            }
   556	            Integer corOrigem = cores.get(a.getOrigem());
   557	            Integer corDestino = cores.get(a.getDestino());
   558	            if (corOrigem == null || corDestino == null || corOrigem.equals(corDestino)) {
   559	                return false;
   560	            }
   561	        }
   562	        return true;
   563	    }

   569	    /**
   570	     Algoritmo A* entre dois vértices.
   571	
   572	     Cada vértice é avaliado por f(n) = g(n) + h(n), onde g(n) é o custo real
   573	     já percorrido desde a origem, somando os pesos das arestas, e h(n) é a
   574	     estimativa do que falta até o destino, calculada pela fórmula de
   575	     Haversine sobre as coordenadas geográficas.
   576	
   577	     Como Haversine mede em linha reta, h(n) nunca superestima o que falta,
   578	     ou seja, a heurística é admissível, e por isso o A* devolve o caminho de
   579	     menor custo. Em grafo sem coordenadas h(n) vale zero e o algoritmo se
   580	     comporta como o de Dijkstra.
   581	
   582	     Em grafo dirigido a lista de adjacência contém apenas os arcos que saem
   583	     do vértice, então o sentido é respeitado.
   584	     */
   585	    public ResultadoCaminho aEstrela(String idOrigem, String idDestino) {
   586	        Vertice origem = buscarVertice(idOrigem);
   587	        Vertice destino = buscarVertice(idDestino);
   588	
   589	        if (origem == null) {
   590	            throw new IllegalArgumentException("Vértice de origem não existe: " + idOrigem);
   591	        }
   592	        if (destino == null) {
   593	            throw new IllegalArgumentException("Vértice de destino não existe: " + idDestino);
   594	        }
   595	
   596	        Map<Vertice, Double> custoConhecido = new HashMap<>();
   597	        Map<Vertice, Aresta> chegada = new HashMap<>();
   598	
   599	        // Lista fechada: vértices já resolvidos. Como eles entram aqui na
   600	        // mesma ordem em que saem da fila, esta lista também é a ordem de
   601	        // expansão mostrada na tela.
   602	        List<Vertice> fechados = new ArrayList<>();
   603	
   604	        // Lista aberta: a fronteira, ordenada por f(n) = g(n) + h(n).
   605	        PriorityQueue<NoAberto> abertos = new PriorityQueue<>(
   606	                Comparator.comparingDouble((NoAberto n) -> n.f).thenComparing(n -> n.vertice.getId()));
   607	
   608	        custoConhecido.put(origem, 0.0);
   609	        abertos.add(new NoAberto(origem, origem.distanciaHaversine(destino)));
   610	
   611	        boolean encontrou = false;
   612	
   613	        while (!abertos.isEmpty()) {
   614	            Vertice atual = abertos.poll().vertice;
   615	
   616	            // O mesmo vértice pode ter sido enfileirado mais de uma vez com
   617	            // estimativas diferentes; a entrada antiga é descartada.
   618	            if (fechados.contains(atual)) {
   619	                continue;
   620	            }
   621	            fechados.add(atual);
   622	
   623	            if (atual == destino) {
   624	                encontrou = true;
   625	                break;
   626	            }
   627	
   628	            for (Aresta a : atual.getAdjacencias()) {
   629	                Vertice vizinho = a.oposto(atual);
   630	
   631	                if (fechados.contains(vizinho)) {
   632	                    continue;
   633	                }
   634	
   635	                double novoCusto = custoConhecido.get(atual) + a.getPeso();
   636	
   637	                // Só vale a pena seguir por aqui se for mais barato que o
   638	                // melhor caminho já conhecido até este vizinho.
   639	                if (!custoConhecido.containsKey(vizinho) || novoCusto < custoConhecido.get(vizinho)) {
   640	                    custoConhecido.put(vizinho, novoCusto);
   641	                    chegada.put(vizinho, a);
   642	                    abertos.add(new NoAberto(vizinho, novoCusto + vizinho.distanciaHaversine(destino)));
   643	                }
   644	            }
   645	        }
   646	
   647	        List<Vertice> caminhoVertices = new ArrayList<>();
   648	        List<Aresta> caminhoArestas = new ArrayList<>();
   649	
   650	        if (encontrou) {
   651	            Vertice atual = destino;
   652	            caminhoVertices.add(atual);
   653	            while (chegada.containsKey(atual)) {
   654	                Aresta a = chegada.get(atual);
   655	                caminhoArestas.add(0, a);
   656	                atual = a.oposto(atual);
   657	                caminhoVertices.add(0, atual);
   658	            }
   659	        }
   660	
   661	        double custoTotal = encontrou ? custoConhecido.get(destino) : 0.0;
   662	
   663	        return new ResultadoCaminho(origem, destino, encontrou, caminhoVertices,
   664	                caminhoArestas, custoTotal, fechados);
   665	    }

   700	    /**
   701	     * Matriz de alcançabilidade (fecho transitivo) pelo algoritmo de Roy,
   702	     * também conhecido como Roy-Warshall, aplicado sobre a matriz booleana
   703	     * de adjacência.
   704	     */
   705	    public boolean[][] matrizAlcancabilidade() {
   706	        int n = vertices.size();
   707	        boolean[][] r = new boolean[n][n];
   708	
   709	        // Todo vértice alcança a si mesmo.
   710	        for (int i = 0; i < n; i++) {
   711	            r[i][i] = true;
   712	        }
   713	
   714	        // A posição do vértice na lista já é o seu índice na matriz.
   715	        for (Aresta a : arestas) {
   716	            int i = vertices.indexOf(a.getOrigem());
   717	            int j = vertices.indexOf(a.getDestino());
   718	            r[i][j] = true;
   719	            if (!dirigido) {
   720	                r[j][i] = true;
   721	            }
   722	        }
   723	
   724	        // Fecho transitivo: se i alcança k e k alcança j, então i alcança j.
   725	        for (int k = 0; k < n; k++) {
   726	            for (int i = 0; i < n; i++) {
   727	                if (!r[i][k]) {
   728	                    continue;
   729	                }
   730	                for (int j = 0; j < n; j++) {
   731	                    if (r[k][j]) {
   732	                        r[i][j] = true;
   733	                    }
   734	                }
   735	            }
   736	        }
   737	        return r;
   738	    }

   740	    /**
   741	     * Componentes conexas (grafo não dirigido) ou fortemente conexas
   742	     * (grafo dirigido): v e w pertencem ao mesmo conjunto quando
   743	     * v alcança w e w alcança v na matriz de alcançabilidade.
   744	     */
   745	    public List<List<Vertice>> royComponentes() {
   746	        boolean[][] r = matrizAlcancabilidade();
   747	        int n = vertices.size();
   748	        boolean[] agrupado = new boolean[n];
   749	        List<List<Vertice>> componentes = new ArrayList<>();
   750	
   751	        for (int i = 0; i < n; i++) {
   752	            if (agrupado[i]) {
   753	                continue;
   754	            }
   755	            List<Vertice> componente = new ArrayList<>();
   756	            for (int j = 0; j < n; j++) {
   757	                if (!agrupado[j] && r[i][j] && r[j][i]) {
   758	                    agrupado[j] = true;
   759	                    componente.add(vertices.get(j));
   760	                }
   761	            }
   762	            componentes.add(componente);
   763	        }
   764	        return componentes;
   765	    }

    11	/**
    12	 Leitura e gravação de grafo em arquivo de texto.
    13	
    14	 Formato, uma linha por item, campos separados por ponto e vírgula:
    15	
    16	 <pre>
    17	 # linhas iniciadas por # são comentário e linhas em branco são ignoradas
    18	 NAO_DIRIGIDO                           primeira linha útil: tipo do grafo
    19	 V;Curitiba;-25.4133;-49.2679           vértice com coordenada geográfica
    20	 V;Toledo                               vértice sem coordenada
    21	 A;e1;Curitiba;Toledo;114               aresta: id, origem, destino, peso
    22	 </pre>
    23	
    24	 A coordenada é opcional: vértice sem ela funciona normalmente, apenas não
    25	 contribui com heurística para o A*.
    26	 */
    27	public class ArquivoGrafo {
    28	
    29	    private ArquivoGrafo() {
    30	    }
    31	
    32	    /** Lê um grafo do arquivo. Indica a linha quando encontra erro de formato. */
    33	    public static Grafo carregar(File arquivo) throws IOException {
    34	        Grafo grafo = null;
    35	        int numeroLinha = 0;
    36	
    37	        try (BufferedReader leitor = new BufferedReader(
    38	                new InputStreamReader(new FileInputStream(arquivo), StandardCharsets.UTF_8))) {
    39	
    40	            String linha;
    41	            while ((linha = leitor.readLine()) != null) {
    42	                numeroLinha++;
    43	                linha = linha.trim();
    44	
    45	                if (linha.isEmpty() || linha.startsWith("#")) {
    46	                    continue;
    47	                }
    48	
    49	                String[] campos = linha.split(";");
    50	                for (int i = 0; i < campos.length; i++) {
    51	                    campos[i] = campos[i].trim();
    52	                }
    53	
    54	                try {
    55	                    if (grafo == null) {
    56	                        grafo = new Grafo(lerTipo(campos[0]));
    57	                    } else {
    58	                        lerItem(grafo, campos);
    59	                    }
    60	                } catch (RuntimeException ex) {
    61	                    throw new IOException("Linha " + numeroLinha + ": " + ex.getMessage());
    62	                }
    63	            }
    64	        }
    65	
    66	        if (grafo == null) {
    67	            throw new IOException("Arquivo sem conteúdo: falta a linha com o tipo do grafo.");
    68	        }
    69	        return grafo;
    70	    }
    71	
    72	    /** Grava o grafo no arquivo, no mesmo formato que o carregar espera. */
    73	    public static void salvar(Grafo grafo, File arquivo) throws IOException {
    74	        try (BufferedWriter escritor = new BufferedWriter(
    75	                new OutputStreamWriter(new FileOutputStream(arquivo), StandardCharsets.UTF_8))) {
    76	
    77	            escritor.write("# Grafo gravado pelo programa");
    78	            escritor.newLine();
    79	            escritor.write(grafo.isDirigido() ? "DIRIGIDO" : "NAO_DIRIGIDO");
    80	            escritor.newLine();
    81	
    82	            for (Vertice v : grafo.getVertices()) {
    83	                if (v.temCoordenadas()) {
    84	                    escritor.write("V;" + v.getId() + ";" + v.getLatitude() + ";" + v.getLongitude());
    85	                } else {
    86	                    escritor.write("V;" + v.getId());
    87	                }
    88	                escritor.newLine();
    89	            }
    90	
    91	            for (Aresta a : grafo.getArestas()) {
    92	                escritor.write("A;" + a.getId() + ";" + a.getOrigem().getId()
    93	                        + ";" + a.getDestino().getId() + ";" + a.getPesoFormatado());
    94	                escritor.newLine();
    95	            }
    96	        }
    97	    }

    99	    private static boolean lerTipo(String texto) {
   100	        if (texto.equalsIgnoreCase("DIRIGIDO")) {
   101	            return true;
   102	        }
   103	        if (texto.equalsIgnoreCase("NAO_DIRIGIDO") || texto.equalsIgnoreCase("NÃO_DIRIGIDO")) {
   104	            return false;
   105	        }
   106	        throw new IllegalArgumentException(
   107	                "a primeira linha deve ser DIRIGIDO ou NAO_DIRIGIDO, e veio: " + texto);
   108	    }
   109	
   110	    private static void lerItem(Grafo grafo, String[] campos) {
   111	        if (campos[0].equalsIgnoreCase("V")) {
   112	            if (campos.length == 2) {
   113	                grafo.inserirVertice(campos[1]);
   114	            } else if (campos.length == 4) {
   115	                grafo.inserirVertice(campos[1], lerNumero(campos[2]), lerNumero(campos[3]));
   116	            } else {
   117	                throw new IllegalArgumentException(
   118	                        "vértice deve ser V;id ou V;id;latitude;longitude");
   119	            }
   120	
   121	        } else if (campos[0].equalsIgnoreCase("A")) {
   122	            if (campos.length != 5) {
   123	                throw new IllegalArgumentException(
   124	                        "aresta/arco deve ser A;id;origem;destino;peso");
   125	            }
   126	            grafo.inserirAresta(campos[1], campos[2], campos[3], lerNumero(campos[4]));
   127	
   128	        } else {
   129	            throw new IllegalArgumentException(
   130	                    "a linha deve começar com V de vértice ou A de aresta, e veio: " + campos[0]);
   131	        }
   132	    }
   133	
   134	    private static double lerNumero(String texto) {
   135	        try {
   136	            return Double.parseDouble(texto.replace(',', '.'));
   137	        } catch (NumberFormatException ex) {
   138	            throw new IllegalArgumentException("valor numérico inválido: " + texto);
   139	        }
   140	    }

    95	    private void posicionarEmCirculo() {
    96	        List<Vertice> lista = grafo.getVertices();
    97	        int n = lista.size();
    98	        int cx = getWidth() / 2;
    99	        int cy = getHeight() / 2;
   100	        int raio = Math.max(60, Math.min(getWidth(), getHeight()) / 2 - 60);
   101	
   102	        posicoes.clear();
   103	        for (int i = 0; i < n; i++) {
   104	            double angulo = -Math.PI / 2 + (2 * Math.PI * i) / n;
   105	            posicoes.put(lista.get(i).getId(),
   106	                    new Point((int) (cx + raio * Math.cos(angulo)), (int) (cy + raio * Math.sin(angulo))));
   107	        }

   110	    private void desenharAresta(Graphics2D g2, Aresta a) {
   111	        Point p1 = posicoes.get(a.getOrigem().getId());
   112	        Point p2 = posicoes.get(a.getDestino().getId());
   113	
   114	        boolean destaque = destacadas.contains(a.getId());
   115	        g2.setColor(destaque ? COR_DESTAQUE : Color.DARK_GRAY);
   116	        g2.setStroke(new BasicStroke(destaque ? 3f : 1.5f));
   117	
   118	        String texto = a.getId() + " (" + a.getPesoFormatado() + ")";
   119	
   120	        if (a.isLaco()) {
   121	            g2.drawOval(p1.x - 18, p1.y - RAIO - 26, 36, 30);
   122	            escrever(g2, texto, p1.x, p1.y - RAIO - 32);
   123	            return;
   124	        }
   125	
   126	        double dx = p2.x - p1.x;
   127	        double dy = p2.y - p1.y;
   128	        double dist = Math.hypot(dx, dy);
   129	        double ux = dx / dist;
   130	        double uy = dy / dist;
   131	
   132	        // Em grafo dirigido a reta é deslocada para a lateral. Como a
   133	        // perpendicular acompanha o sentido origem -> destino, o arco de ida e
   134	        // o de volta entre o mesmo par caem em lados opostos automaticamente.
   135	        int desvio = grafo.isDirigido() ? 16 : 0;
   136	        int deslocX = (int) (-uy * desvio);
   137	        int deslocY = (int) (ux * desvio);
   138	
   139	        // A reta começa e termina na borda dos círculos, não no centro.
   140	        int x1 = (int) (p1.x + ux * RAIO) + deslocX;
   141	        int y1 = (int) (p1.y + uy * RAIO) + deslocY;
   142	        int x2 = (int) (p2.x - ux * RAIO) + deslocX;
   143	        int y2 = (int) (p2.y - uy * RAIO) + deslocY;
   144	
   145	        g2.drawLine(x1, y1, x2, y2);
   146	
   147	        if (grafo.isDirigido()) {
   148	            desenharSeta(g2, x2, y2, ux, uy);
   149	        }
   150	
   151	        escrever(g2, texto, (x1 + x2) / 2, (y1 + y2) / 2);
   152	    }

   154	    private void desenharSeta(Graphics2D g2, int x, int y, double ux, double uy) {
   155	        double angulo = Math.atan2(uy, ux);
   156	        double tamanho = 12;
   157	        double abertura = Math.toRadians(22);
   158	
   159	        int[] xs = {x, (int) (x - tamanho * Math.cos(angulo - abertura)),
   160	                (int) (x - tamanho * Math.cos(angulo + abertura))};
   161	        int[] ys = {y, (int) (y - tamanho * Math.sin(angulo - abertura)),
   162	                (int) (y - tamanho * Math.sin(angulo + abertura))};
   163	
   164	        g2.fillPolygon(xs, ys, 3);
   165	    }

   175	        g2.setColor(cor);

   384	        String origem = perguntar("Vértice de saída:");
   385	        if (origem == null) {
   386	            return;
   387	        }
   388	        String destino = perguntar("Vértice de chegada:");
   389	        if (destino == null) {
   390	            return;
   391	        }
   392	        try {

   394	
   395	            String texto = "===== BUSCA EM PROFUNDIDADE GUIADA =====\n";
   396	            texto += "Saída: " + b.getOrigem().getId()
   397	                    + "   Chegada: " + b.getDestino().getId() + "\n";
   398	            texto += "Ordem de visita: " + b.getOrdemVisita() + "\n";
   399	            texto += "Arestas da árvore:\n";
   400	            for (Aresta a : b.getArestasArvore()) {
   401	                texto += "  " + a.getId() + ": " + a.getOrigem().getId()
   402	                        + (grafo.isDirigido() ? " -> " : " -- ") + a.getDestino().getId()
   403	                        + "  peso " + a.getPesoFormatado() + "\n";

   405	            if (b.isEncontrou()) {
   406	                texto += "Caminho: " + b.getCaminhoVertices()
   407	                        + "   custo " + Aresta.formatarPeso(b.getCustoCaminho());

   413	
   414	            Map<String, Color> cores = new HashMap<>();
   415	            cores.put(b.getOrigem().getId(), new Color(0xA8, 0xE0, 0xB0));
   416	            cores.put(b.getDestino().getId(), new Color(0xF5, 0xB4, 0xA8));
   417	            painel.realcar(b.getArestasArvore(), idsDe(b.getCaminhoArestas()), cores);
   418	        } catch (RuntimeException ex) {
   419	            erro(ex);
   420	        }
   421	    }

   449	        String origem = perguntar("Cidade de origem:");
   450	        if (origem == null) {
   451	            return;
   452	        }
   453	        String destino = perguntar("Cidade de destino:");
   454	        if (destino == null) {
   455	            return;
   456	        }
   457	        try {
   458	            Grafo.ResultadoCaminho r = grafo.aEstrela(origem, destino);

   482	                texto += "CUSTO TOTAL: " + Aresta.formatarPeso(r.getCustoTotal()) + " km";
   483	            } else {

   485	                        + r.getOrigem().getId() + ".";
   486	            }
   487	            imprimir(texto);
   488	
   489	            Map<String, Color> cores = new HashMap<>();
